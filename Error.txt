using MES.Net.Application.Services.Print;
using MES.Net.Shared.DTOs.Print;
using MES.Net.Web.Filters;
using System;
using System.Threading.Tasks;
using System.Web.Http;

namespace MES.Net.Web.Controllers.Print
{
    [RoutePrefix("api/print/print-label")]
    public class PrintLabelController : ApiController
    {
        private readonly IPrintLabelService _printLabelService;

        public PrintLabelController()
        {
            _printLabelService = new PrintLabelService();
        }
        public PrintLabelController(IPrintLabelService printLabelService)
        {
            _printLabelService = printLabelService;
        }

        /// <summary>
        /// 取得 ISG 出貨單報表資料
        /// </summary>
        [HttpPost, Route("shipNo")]
        public async Task<IHttpActionResult> GetISGData([FromBody] PrintISGRequest request)
        {
            try
            {
                // 防呆：確保前端有傳值過來
                if (request == null || string.IsNullOrWhiteSpace(request.ShipNo))
                {
                    return Ok(new { Success = false, Message = "請輸入出貨單號! (Please key in ShipNo!)" });
                }

                var data = await _printLabelService.GetInitDataAsync();
                return Ok(new { Success = true, Data = data });
            }
            catch (Exception ex)
            {
                return Ok(new { Success = false, Message = ex.Message });
            }
        }
    }
}
