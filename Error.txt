    '----

    ' Action

Sub Form_Load()
    
On Error GoTo ExitHandler:
Dim sProcID As String
Dim typErrInfo As tErrInfo
Dim sRawSqlReplyboxName As String
Dim sRawSqlIniFilename  As String
Dim oFSO As Object
Dim bIsCancelled As Boolean
Dim sErrMsg As String
Dim lCompNameLen As Long
Dim sCompName    As String
Dim vRtn         As Variant
Dim sMsg As String
Dim Init         As Boolean

'----
' Init
'----
    Init = False
    ' initialize moAppLog
    If moAppLog Is Nothing Then
        Set moAppLog = New clsLogTraceMsg
        moAppLog.ClientID = sCompName & Format(Time, "hhmmss")
        moAppLog.Init App.Path, , App.Title, , , , , , , True, , , , , , True
        moAppLog.LogLevel = glLOG_TRACE 'Val(msLogLevel)
    End If
    
    sProcID = "Form_Load"
    Call LogProcIn(msMODULE_ID, sProcID, moAppLog)
    'Me.Show
        
    msRuleName = "ReturnToDispatch"
    'Add by Robin at 2017/10/18, initial for 太久沒 login 重做一次時
    Set oFSO = Nothing
    Set moProRawSql = Nothing
    Set moRawSql = Nothing
'----
' Condition Checking
'----
    ' check if the existing instance is running
'     If App.PrevInstance Then
'         Call RaiseError(glERR_APPISRUNNING, _
'                     FormatErrorText(gsETX_APPISRUNNING, App.Title))
'     End If

    ' Check if the INI file is specified in the Command
     msIniFile = vbNullString
     If UtGetCmdToken(msCMD_INIFILE, msIniFile) Then
     ' inifile is specified in command
         If Len(msIniFile) = 0 Then
             Call RaiseError(glERR_INVALIDVALUE, _
                         FormatErrorText(gsETX_INVALIDVALUE, "msIniFile::String", vbNullString))
         End If
     Else
     ' take the default ini file
         msIniFile = msDEF_INIFILE
     End If

    ' check if the ini file is valid
    On Error Resume Next
    Set oFSO = CreateObject("Scripting.FileSystemObject") ' New FileSystemObject
    If Err Then Err.Clear
    On Error GoTo ExitHandler
    If oFSO Is Nothing Then
        Call RaiseError(glERR_INVALIDOBJECT, _
                    FormatErrorText(gsETX_INVALIDOBJECT, "oFSO:FileSystemObject"))
    End If
    If Not oFSO.FileExists(msIniFile) Then
    ' ini file is missing
        Call mCreateDefIni(msIniFile, moAppLog)
    End If
'----
' Action
'----

    msFormTitle = "SortRun LotStart"
    
    'Get the hostname, mbxname, servername from ini file
    Call modWinAPI.ApIniGetItem(msIniFile, msSECT_MBXINFO, msITEM_HOSTNAME, msHostName, msDEF_HOSTNAME, moAppLog)
    Call modWinAPI.ApIniGetItem(msIniFile, msSECT_MBXINFO, msITEM_MBXNAME, msMbxName, msDEF_MBXNAME, moAppLog)
    Call modWinAPI.ApIniGetItem(msIniFile, msSECT_MBXINFO, msITEM_SRVNAME, msSrvName, msDEF_SRVNAME, moAppLog)
    Call modWinAPI.ApIniGetItem(msIniFile, msSECT_MBXINFO, msITEM_AUTORECONNECTION, msAUTORECONNECTION, msDEF_AUTORECONNECTION, moAppLog)
    
    Call modWinAPI.ApIniGetItem(msIniFile, msSECT_LOGFILE, msITEM_Level, msLogLevel, "3", moAppLog)
    Call modWinAPI.ApIniWriteItem(msIniFile, msSECT_LOGFILE, msITEM_Level, msLogLevel, moAppLog)
    
    lCompNameLen = 256
    sCompName = Space(lCompNameLen)
    vRtn = modWinAPI.ApGetComputerName(sCompName)
    If vRtn = True Then
        sRawSqlReplyboxName = sCompName & Format(Time, "hhmmss") & ".ProRawSQL"
    End If
    
    
    'Show frmMBXInfo form to let user to input mbx information
    Load frmMBXInfo
    With frmMBXInfo
        .ShowForm msHostName, msMbxName, msSrvName, msAUTORECONNECTION
        .Show vbModal
        If Not .IsCancelled Then
            Call modWinAPI.ApIniWriteItem(msIniFile, msSECT_MBXINFO, msITEM_HOSTNAME, .HostName, moAppLog)
            Call modWinAPI.ApIniWriteItem(msIniFile, msSECT_MBXINFO, msITEM_MBXNAME, .MbxName, moAppLog)
            Call modWinAPI.ApIniWriteItem(msIniFile, msSECT_MBXINFO, msITEM_SRVNAME, .PCMBXSrvName, moAppLog)
        Else
            Call RaiseError(glERR_USERCANCELLED, _
                        FormatErrorText(gsETX_USERCANCELLED, "Application Initaization (MBX Connection)..."))
            GoTo ExitHandler
        End If

        If mInitFwCtrls(.HostName, .MbxName, .PCMBXSrvName, moAppLog) = False Then
            Unload frmMBXInfo
            Init = False
            Call Restart
            
        End If
        
        Unload frmMBXInfo
    End With
  
        
    sRawSqlIniFilename = "C:\Rawsql.ini"
    Set moProRawSql = Me.ProRawSQL1
    moProRawSql.Initialize moCwMbx, _
                           sRawSqlReplyboxName, _
                           sRawSqlIniFilename
        
    Set moRawSql = New clsRawSql
    Set moRawSql.CwMbx = moCwMbx
    Set moRawSql.AppLog = moAppLog
    moRawSql.RawSqlMailBox = sCompName & Format(Time, "hhmmss") & ".RawSql_Reply"
        
    ' initialize the menu control
    MousePointer = vbDefault
    
    cmdLogin.Enabled = True
    cmdLogout.Enabled = False
    
    Timer1.Interval = 60000 '60 sec
    Timer1.Enabled = True
    
    mlTimer = 0
    
    Init = True
    
    If msAUTORECONNECTION = "1" Then
        '監控資料庫是否連線
        Timer2.Interval = 60000
        Timer2.Enabled = True
    End If
    
'    cmdTest.Visible = True
    
'----
' Done
'----
    Call UtFormatProcID(sProcID)
    
ExitHandler:
     Call GetErrInfo(msMODULE_ID, sProcID, typErrInfo, Erl)
     Call LogProcOut(msMODULE_ID, sProcID, typErrInfo, moAppLog)
    ' cleaning up
     Set oFSO = Nothing
    If Not frmMBXInfo Is Nothing And Init = False Then
        Unload frmMBXInfo
        Unload Me
    End If

ErrorHandler:
    If typErrInfo.lErrNumber Then
        ' NOTE 2:
        ' If you have custom handling of some Errors, please
        ' UN-REMARED the following Select Case block!
        ' Also, modify if neccessarily!!!
        '---- Start of Select Case Block ----
        Select Case typErrInfo.lErrNumber
            Case glERR_USERCANCELLED
                    typErrInfo.sUserText = "You have cancelled the mbx connection!  Application will terminate now..."
            Case glERR_APPISRUNNING
                     'Call UtShowMsgBox("Application is already running..." & vbCrLf & _
                                       "Application will not start!" & vbCrLf & typErrInfo.sErrDescription, vbExclamation + vbOKOnly, App.Title, False)
            Case Else
        End Select
        MousePointer = vbDefault
        Unload Me
        On Error GoTo ExitHandler:
        Call HandleError(False, typErrInfo, , moAppLog, False)
        Call Restart
    End If
End Sub

Private Function mInitFwCtrls(ByRef sHostName As String, ByRef sMbxName As String, ByRef sSrvName As String, Optional ByRef oLogCtrl As clsLogTraceMsg) As Boolean
    On Error GoTo ExitHandler:
    Dim sProcID As String
    Dim typErrInfo As tErrInfo
    Dim lStartTick As Long
    Dim mlPrevPointer As Long

'----
' Init
'----
    sProcID = "mInitFwCtrls"
    Call LogProcIn(msMODULE_ID, sProcID, oLogCtrl) '
    mInitFwCtrls = True
    MousePointer = VBRUN.MousePointerConstants.vbHourglass
'----
' Condition Checking
'----
    ' <Put your condition checking codes here>...
    If Len(sHostName) = 0 Then
        Call RaiseError(glERR_INVALIDVALUE, _
                FormatErrorText(gsETX_INVALIDVALUE, "sHostName::String", vbNullString))
    End If
    
    If Len(sSrvName) = 0 Then
        Call RaiseError(glERR_INVALIDVALUE, _
                FormatErrorText(gsETX_INVALIDVALUE, "sSrvName::String", vbNullString))
    End If
    
    If Len(sMbxName) = 0 Then
        Call RaiseError(glERR_INVALIDVALUE, _
                FormatErrorText(gsETX_INVALIDVALUE, "sMbxName::String", vbNullString))
    End If
        
'----
' Action
'----
    'On Error Resume Next
    'Initialize the CwMbx ctrl
    
    Set moCwMbx = Me.CwMbxMain
    Set moFwOPR = Me.FwOPRMain
    Set moFwPRP = Me.FwPRPMain
    Set moFwWIP = Me.FwWIPMain
    Set moFwWF = Me.FwWFMain
    Set moFwMDL = Me.FwMDLMain
    Set moFwCAT = Me.FwCATMain
    
    With moCwMbx ' Me.CwMbxMain
        If .Status = Connected Then
            .Disconnect
        End If
        Call LogTransBegin(msMODULE_ID, sProcID, "MBX.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
        .HostName = sHostName
        .MbxName = sMbxName
        .ServerName = sSrvName
        Call .GetError(typErrInfo.lErrNumber, typErrInfo.sErrDescription)
        Call LogTransEnd(msMODULE_ID, sProcID, "MBX.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
    End With
    
    ' Initialize the FwOPR ctrl
    
    With moFwOPR ' Me.FwOPRMain
On Error Resume Next
            Call LogTransBegin(msMODULE_ID, sProcID, "OPR.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
            Call .Initialize(sHostName, sMbxName, sSrvName)
            Call LogTransEnd(msMODULE_ID, sProcID, "OPR.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
On Error GoTo ExitHandler:
    End With

'    ' Initialize the FwWF ctrl
    
    With moFwWF ' Me.FwWFMain
On Error Resume Next
            Call LogTransBegin(msMODULE_ID, sProcID, "WF.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
            Call .Initialize(sHostName, sMbxName, sSrvName)
            Call LogTransEnd(msMODULE_ID, sProcID, "WF.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
On Error GoTo ExitHandler:
    End With

    ' Initialize the FwPRP ctrl
    With moFwPRP ' Me.FwPRPMain
On Error Resume Next
        Call LogTransBegin(msMODULE_ID, sProcID, "PRP.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
        Call .Initialize(sHostName, sMbxName, sSrvName)
        Call LogTransEnd(msMODULE_ID, sProcID, "PRP.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
On Error GoTo ExitHandler:

On Error Resume Next
            Call LogTransBegin(msMODULE_ID, sProcID, "WIP.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
            Call .Initialize(sHostName, sMbxName, sSrvName)
            Call LogTransEnd(msMODULE_ID, sProcID, "WIP.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
On Error GoTo ExitHandler:
    End With
    
'    ' Initialize the FwCAT ctrl
    With moFwCAT 'Me.FwCATMain
        Call LogTransBegin(msMODULE_ID, sProcID, "CAT.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)

        Call LogTransEnd(msMODULE_ID, sProcID, "CAT.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
    End With

    ' Initialize the FwMDL ctrl
    With moFwMDL ' Me.FwMDLMain
On Error Resume Next
            Call LogTransBegin(msMODULE_ID, sProcID, "MDL.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)

            Call LogTransEnd(msMODULE_ID, sProcID, "MDL.Init", moAppLog, lStartTick, "MbxName:=" & sMbxName, "HostName:=" & sHostName, "PCMBXsrvName:=" & sSrvName)
On Error GoTo ExitHandler:
    End With
    

    If Err.Number = 4170 Then
        Err.Clear
    End If
 
'----
' Done
'----

ExitHandler:
    ' NOTE 1:
    ' MUST CALL GetErrInfo() here first before another action
    Call GetErrInfo(msMODULE_ID, sProcID, typErrInfo, Erl)
    Call LogProcOut(msMODULE_ID, sProcID, typErrInfo, oLogCtrl)
    ' <Your cleaning up codes goes here...>
    
ErrorHandler:

        ' NOTE 2:
        ' If you have custom handling of some Errors, please
        ' UN-REMARED the following Select Case block!
        ' Also, modify if neccessarily!!!
        '---- Start of Select Case Block ----
        'Select Case typErrInfo.lErrNumber
        '   Case glERR_MBX_CONNECTION_Error
        '       ' Retry code goes here...
        '   Case Else
        '       ' ... etc ...
        'End Select
        '---- Start of Select Case Block ----
'        Call UtShowMsgBox("Error occurred while trying to establish MBX connection!" & vbCrLf & _
'                "Error (" & typErrInfo.lErrNumber & ") " & typErrInfo.sErrDescription, vbCritical + vbOKOnly, App.Title, False)

End Function

Private Sub Form_Unload(Cancel As Integer)

Dim pid As Long
Dim hProcess As Long

Dim r          As Long

Dim sExeName As String

    If Not moFwOPR.ActiveUser Is Nothing Then moFwOPR.Logout
End If

Call ResetFwControls

bFound = False

lSnapShot = CreateToolhelpSnapshot(TH32CS_SNAPPROCESS, 0&)
If lSnapShot <> 0 Then
    uProcess.lSize = Len(uProcess)
    r = ProcessFirst(lSnapShot, uProcess)

    Do While r
        sExeName = Left(uProcess.sExeFile, InStr(1, uProcess.sExeFile, vbNullChar) - 1)
        pid = uProcess.lProcessId
        
        If UCase(sExeName) Like UCase("ReturnToDispatch*") Then
           bFound = True
           Exit Do
        End If
        r = ProcessNext(lSnapShot, uProcess)
    Loop
    
    CloseHandle (lSnapShot)
End If

If bFound = True Then
    'pid = ApiFindWindow(vbNullString, App.EXEName) 'Shell(App.Path & "\" & App.EXEName, vbNormalFocus)
       
    hProcess = OpenProcess(&H1, 0, pid)
    TerminateProcess hProcess, 0
    CloseHandle hProcess
End If
End

'----

    Set oLot = moFwWIP.LotById(sLotID)

    ElseIf oEquipment.State = gsEQPSTATUS_WSETUP Or _

        End If

        oEquipment.Refresh
        
        sEqpH1 = oEquipment.CustomAttributes(gsEQP_CUSTOMATTR_SUBSYS1)

            oEqpH1.Refresh
        End If
        
        sEqpH2 = oEquipment.CustomAttributes(gsEQP_CUSTOMATTR_SUBSYS2)
        If Len(Trim(sEqpH2)) <> 0 Then
            Set oEqpH2 = FwuRetrieveEqp(moFwMDL, sEqpH2)

                                             sUserName, _
                                             sGroupHistory, _

    Set oComment = moFwWIP.CreateFwComment

    'CurStepNo

    End If
    
    Set colWaitRProbeCards = New Collection
    Set colReqProbeCards = New Collection
    
    bTECNLotSpec = CheckTECNLotInfo(moAppLog, moFwWIP, moFwWF, moCwMbx, oLot.Id)
    
    If bTECNLotSpec = True Then
        Call GetReqProbeCardsbyLotSpec(oLot, _
                                       oEquipment.CustomAttributes(modConstFwAttr.gsEQP_CUSTOMATTR_EQ_TYPE2).Value, _
                                       "", _
                                       colReqProbeCards)
    Else
        Call GetReqProbeCards(oLot, _
                              oEquipment.CustomAttributes(modConstFwAttr.gsEQP_CUSTOMATTR_EQ_TYPE2).Value, _
                              colReqProbeCards)
    End If
    
    sSQL = "Select A." & gsCAT_TEI_PROBECARD_ID & " from " & gsCAT_TBL_EQ_INFO & " A " & _
           " where A." & gsCAT_TEI_EQ_ID & "='" & oEquipment.Id & "'"
    Set colRS = moProRawSql.QueryDatabase(sSQL)
    If colRS.Count > 0 Then
        If InStr(colRS.Item(1).Item(1), "-") > 0 Then
            vCurrProbeCard = Left(colRS.Item(1).Item(1), InStr(colRS.Item(1).Item(1), "-") - 1)
        Else
            vCurrProbeCard = colRS.Item(1).Item(1)
        End If
    End If
            
    Call GetWaitRProbeCards(sEqID, oLot.Id, colWaitRProbeCards)
             
    For Each vReqProbeCard In colReqProbeCards
    
        bFound = False
        
        For Each vWaitRProbeCard In colWaitRProbeCards
            If Trim(vReqProbeCard) = Trim(vWaitRProbeCard) Then '
                bFound = True
            End If
        Next vWaitRProbeCard
        
        If bFound = False And Trim(vReqProbeCard) <> Trim(vCurrProbeCard) Then
              Call UpdateAccReserveQty(CStr(vReqProbeCard))
        End If
        
    Next vReqProbeCard
                
    Call RepositionToLastRule(moFwPRP, _
                              moFwWF, _
                              oLot, _
                              oLot.CurrentStep.Steps.Item(1).CurrentRule.Id, _
                              sUserName, _
                              sGroupHistory, _

'----

        End If

                'Get current handler type2
                sCurHandlerType2 = oCurHandler.CustomAttributes(modConstFwAttr.gsEQP_CUSTOMATTR_EQ_TYPE2).Value
            End If
        End If
        
        Set oLot = FwuRetrieveLot(moFwWIP, sLotID, moAppLog)
        '不在SORTRUN OR 系統hold
        If Not (UCase(oLot.CurrentStep.Steps.Item(1).CurrentRule.Id) = UCase("SortRun")) Or _
           oLot.Status = gsLOT_STATUS_HOLD Then
            GoTo NextLot
        End If
        
        If colSpec.Count > 0 And lUntestWaferQty > 0 Then
            sType = colSpec.Item(1).Item("type")
            sEqType2 = colSpec.Item(1).Item("eqtype2")
            sPgName = colSpec.Item(1).Item("pgname")
            sPgId = colSpec.Item(1).Item("pgid")
            sPgMode = colSpec.Item(1).Item("pgmode")
            sTemp = colSpec.Item(1).Item("temp")
            sWsDeviceFile = colSpec.Item(1).Item("wsdevicefile")
            sProbecard = colSpec.Item(1).Item("probecardtype")
            sSubsystem = colSpec.Item(1).Item("subsystem")
            sSpecifyEQ = colSpec.Item(1).Item("specifyeq")
            sSpecifyEQIDs = colSpec.Item(1).Item("eqid")
            
            If sType = "PROD" And sLotOwner Like "MFG*" Then
                sSwapPgName = GetSwapPGName(moProRawSql, moAppLog, sLotID, sProdgroup, sStepName, sEqType2, sPgName, False)
                If sSwapPgName <> "" Then
                    If sSwapPgName = "X" Then
                        sComments = "產品(Route, Pgm)變更管理專案:符合程式置換需分批,請通知 PE 協助處"

                                "Lot Cancel Process Start ", rtfTrace, , , sProcID

                                "Lot Cancel Process End ", rtfTrace, , , sProcID
                        Call InsertCpDispatchMonitor(moAppLog, moFwWIP, moFwWF, moCwMbx, moFwMDL, moProRawSql, _
                                                 sLotID, sLotSeq, sEqID, "LotCancel", sProbecard, "SYS", sComments)
                        GoTo NextLot
                    Else
                        sPgName = sSwapPgName
                    End If
                End If
            End If
            
            sProdGroupKey = ""
            sProdCode = ""
            sGrade = ""
            sPGMTECNNo = ""
            sGetPgm = ""
            sGetPgID = ""
            sPGMSource = ""
            sTempTecnNo = ""
            sGetTemp = ""
            sTempSource = ""
            sOverTime = ""
            sTecnLevel = ""
            If sType <> "ENG" Then
                sSQL = "select A." & gsCAT_TIM_PRODGROUPKEY & ", a." & gsCAT_TIM_MASK_OPTION & ", a." & gsCAT_TIM_BE_OPTION & " ,a." & gsCAT_TIM_GRADE & " from " & gsCAT_TBL_IPN_MASTER & " a " & _

                    sGrade = colRS.Item(1).Item(gsCAT_TIM_GRADE)
                    If sStage = "WS" Then
                        sGrade = "C"
                        sProdCode = Left(sIPN, 4) & colRS.Item(1).Item(gsCAT_TIM_MASK_OPTION)
                    ElseIf sStage = "FT" Then
                        sProdCode = Left(sIPN, 4) & colRS.Item(1).Item(gsCAT_TIM_BE_OPTION)
                    End If
                End If
                
                Call modTecn.GetTecnPgmRecipeAttr(moAppLog, moProRawSql, "", sLotID, sIPN, _
                                              sPath, sStepID, sStepName, _
                                              sEqType2, sProdGroupKey, sProdCode, sGrade, "Y", sPgId, sPgName, _
                                              sPGMTECNNo, sGetPgm, sGetPgID, sPGMSource, _
                                              sTempTecnNo, sGetTemp, sTempSource, sOverTime, sTecnLevel)
                If sGetPgm <> "" Then
                    If sGetPgm <> "" Then
                        sPgName = sGetPgm
                        sPgId = sGetPgID
                    End If
                    If sGetTemp <> "" Then
                        sTemp = sGetTemp
                    End If
                End If
            End If
            sStopTestNoList = ""
            bChkStopTest = NotStopTest(sLotID, sIPN, sProdgroup, gsCAT_STOP_TEST_ALL, False, _
                                        sStepName, sEqID, sEqType2, sProbecard, sStopTicNo, _
                                        sSpecifyEQ, sSpecifyEQIDs, sErunTicNO, sPgId, sPgName, sPgMode, _
                                        sTemp, moFwWIP, moAppLog, moProRawSql, sStopTestNoList, sCurHandlerType2, sWsDeviceFile)
            If bChkStopTest = False Then
                If sStopTestNoList <> "" Then
                    sComments = "停測中，停測單號：" & sStopTestNoList & vbNewLine & "Stop Test !!，TicNo: " & sStopTestNoList
                    DispMsg "LotID:" & sLotID & " Time: " & Now & vbNewLine & _
                            "Lot Cancel Process Start ", rtfTrace, , , sProcID
                    Call WSLotCancel(sLotID, sEqID, sComments)
                    DispMsg "LotID:" & sLotID & " Time: " & Now & vbNewLine & _
                            "Lot Cancel Process End ", rtfTrace, , , sProcID
                    Call InsertCpDispatchMonitor(moAppLog, moFwWIP, moFwWF, moCwMbx, moFwMDL, moProRawSql, _
                                             sLotID, sLotSeq, sEqID, "LotCancel", sProbecard, "SYS", sComments)
                    
                    GoTo NextLot
                Else
                    sComments = "停測中，請確認Prod Spec / Lot Spec." & vbNewLine & "Stop Test !!，Please Check Prod Spec / Lot Spec."
                    DispMsg "LotID:" & sLotID & " Time: " & Now & vbNewLine & _
                            "Lot Cancel Process Start ", rtfTrace, , , sProcID
                    Call WSLotCancel(sLotID, sEqID, sComments)
                    DispMsg "LotID:" & sLotID & " Time: " & Now & vbNewLine & _
                            "Lot Cancel Process End ", rtfTrace, , , sProcID
                    Call InsertCpDispatchMonitor(moAppLog, moFwWIP, moFwWF, moCwMbx, moFwMDL, moProRawSql, _
                                             sLotID, sLotSeq, sEqID, "LotCancel", sProbecard, "SYS", sComments)
                    GoTo NextLot
                End If
                
            End If
            
            If sType <> "ENG" Then
                If CheckIPNRelease(sIPN, moFwWIP, moAppLog, moProRawSql) = False Then
                    sComments = "IPN Not Release."
                    DispMsg "LotID:" & sLotID & " Time: " & Now & vbNewLine & _
                            "Lot Cancel Process Start ", rtfTrace, , , sProcID
                    Call WSLotCancel(sLotID, sEqID, sComments)
                    DispMsg "LotID:" & sLotID & " Time: " & Now & vbNewLine & _
                            "Lot Cancel Process End ", rtfTrace, , , sProcID
                    Call InsertCpDispatchMonitor(moAppLog, moFwWIP, moFwWF, moCwMbx, moFwMDL, moProRawSql, _
                                             sLotID, sLotSeq, sEqID, "LotCancel", sProbecard, "SYS", sComments)
                    GoTo NextLot
                End If
            End If
        
        End If
NextLot:
    Next
    
    DispMsg "Time:" & Now & " End...", rtfTrace, , , sProcID
