                /* ── 1. 執行查詢 (對應 ExecQuery) ── */
                async doQuery() {
                    if (!this.qry.lotId.trim()) {
                        VMessageBox.alert('Pls input LotID !!');
                        return;
                    }

                    document.body.style.cursor = 'wait';
                    this.setStatus('QUERYING...', 'info');

                    try {
                        const response = await mesfetch("/api/print/print-setup-form/query", { LotId: this.qry.lotId });

                        if (response.Success) {
                            const data = response.Data;

                            this.formData.lotId = data.LotId || data.lotId;
                            this.formData.stage = data.Stage || data.stage;
                            this.formData.lotStatus = data.LotStatus || data.lotStatus;
                            this.formData.ipn = data.IPN || data.ipn;
                            this.formData.lotOwner = data.LotOwner || data.lotOwner;

                            // 🌟 強制轉換大小寫，確保 Vuetify 絕對能吃到 text 與 value
                            const mapSelectItems = (list) => {
                                if (!list || !Array.isArray(list)) return [];
                                return list.map(item => ({
                                    text: item.Text || item.text || '',
                                    value: item.Value || item.value || ''
                                }));
                            };

                            this.options.stepList = mapSelectItems(data.StepList || data.stepList);
                            this.options.testerIdList = mapSelectItems(data.TesterIdList || data.testerIdList);
                            this.options.subSystemList = mapSelectItems(data.SubSystemList || data.subSystemList);

                            // 🌟 綁定 Setup Reason 下拉選單資料
                            this.options.setupReasonList = mapSelectItems(data.SetupReasonList || data.setupReasonList);

                            this.formData.stepNo = data.SelectedStep || data.selectedStep || '';
                            this.formData.testerId = data.SelectedTesterId || data.selectedTesterId || '';

                            // 🌟 將 API 回傳的 SetupWaferId 綁定到畫面
                            this.formData.setupWaferId1 = data.SetupWaferId1 || data.setupWaferId1 || '';
                            this.formData.setupWaferId2 = data.SetupWaferId2 || data.setupWaferId2 || '';
                            this.formData.setupWaferId3 = data.SetupWaferId3 || data.setupWaferId3 || '';

                            if (this.options.subSystemList.length === 1) {
                                this.formData.subSystemValue = this.options.subSystemList[0].value; // 注意這裡用小寫 value
                                await this.doSubSystemChange();
                            } else {
                                this.formData.subSystemValue = '';
                            }

                            this.setStatus('QUERY SUCCESS', 'success');
                        } else {
                            VMessageBox.alert(response.Message || response.message);
                            this.doClear();
                            this.setStatus('QUERY FAILED', 'error');
                        }
                    } catch (error) {
                        console.error(error);
                        VMessageBox.alert('查詢發生預期外的錯誤', 'Error');
                        this.doClear();
                        this.setStatus('QUERY FAILED', 'error');
                    } finally {
                        document.body.style.cursor = 'default';
                    }
                },
