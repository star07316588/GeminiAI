// 1. 設定目標資料夾路徑與搜尋條件
string folderPath = @"D:\Report\PCDMS";
string searchPattern = "H1B103*";

// 2. 檢查資料夾是否存在，若不存在則建立並直接結束 (因為空資料夾沒檔案可讀)
if (!Directory.Exists(folderPath))
{
    Directory.CreateDirectory(folderPath);
    return; // 或是直接 return / break，視你的外層邏輯而定
}

// 3. 取得符合條件的檔案 (使用 EnumerateFiles 節省記憶體)
var files = Directory.EnumerateFiles(folderPath, searchPattern);

foreach (string file in files)
{
    // 4. 讀取檔案內容
    // 如果 _allDataGET 是 List<string>，建議直接使用 AddRange，效能更好且程式碼更簡潔
    _allDataGET.AddRange(File.ReadLines(file)); 
}
