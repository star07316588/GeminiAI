# 設定存放路徑
$logDir = "C:\Windows\SysWOW64\FwClient\Exe\LogFile\ProcessDetail"

# 確保目錄存在，若無則建立
if (-not (Test-Path $logDir)) {
    New-Item -ItemType Directory -Path $logDir -Force | Out-Null
}

$now = Get-Date
$dateStr = $now.ToString("yyyyMMdd")
$timestamp = $now.ToString("yyyy/MM/dd HH:mm:ss")

# 檔案名稱 (以當天日期命名)
$fileName = "ProcessDetail_${dateStr}.csv"
$filePath = Join-Path -Path $logDir -ChildPath $fileName

# 取得程序詳細資料 (擷取所需的 7 個欄位)
$processData = Get-Process -IncludeUserName -ErrorAction SilentlyContinue | ForEach-Object {
    [PSCustomObject]@{
        名稱 = $_.Name
        PID = $_.Id
        狀態 = if ($_.Responding) { "執行中" } else { "沒有回應" }
        使用者名稱 = if ($_.UserName) { $_.UserName } else { "SYSTEM/Unknown" }
        CPU = if ($_.CPU) { [math]::Round($_.CPU, 2) } else { 0 }
        記憶體 = [math]::Round($_.WorkingSet64 / 1MB, 2).ToString() + " MB"
        紀錄的時間 = $timestamp
    }
}

# 寫入檔案 (若當天檔案已存在則接續寫入，不存在則建立並加上標題欄)
$fileExists = Test-Path $filePath
if ($fileExists) {
    $processData | Export-Csv -Path $filePath -Encoding UTF8 -NoTypeInformation -Append
} else {
    $processData | Export-Csv -Path $filePath -Encoding UTF8 -NoTypeInformation
}

# 清除超過 30 天的舊檔案
$limitDate = (Get-Date).AddDays(-30)
Get-ChildItem -Path $logDir -Filter "ProcessDetail_*.csv" |
    Where-Object { $_.LastWriteTime -lt $limitDate } |
    Remove-Item -Force -ErrorAction SilentlyContinue
